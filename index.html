async function launchWorkspace() { let targetUrl = document.getElementById('urlInput').value.trim(); if (!targetUrl) return; if (!targetUrl.startsWith('http://') && !targetUrl.startsWith('https://')) { if (targetUrl.includes('.') && !targetUrl.includes(' ')) { targetUrl = 'https://' + targetUrl; } else { targetUrl = 'https://www.google.com/search?q=' + encodeURIComponent(targetUrl); } } const launchBtn = document.querySelector('.btn-launch'); if (launchBtn) launchBtn.innerText = 'Connecting...'; if ('serviceWorker' in navigator) { await navigator.serviceWorker.register('/sw.js', { scope: __uv$config.prefix }); // Ensure the Service Worker is active and ready before launching await navigator.serviceWorker.ready; } window.location.href = __uv$config.prefix + __uv$config.encodeUrl(targetUrl); }
